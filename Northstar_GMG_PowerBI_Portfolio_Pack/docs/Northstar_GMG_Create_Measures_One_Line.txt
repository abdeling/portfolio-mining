CREATE MEASURE '_measures'[Opportunity Reduction %] = 0.20
CREATE MEASURE '_measures'[Top 2 Controllable Reasons Hours] = VAR _reasons = CALCULATETABLE ( VALUES ( DimReason[Reason] ), DimReason[ControlClass] = "Controllable" ) RETURN SUMX ( TOPN ( 2, _reasons, [Delay Hours], DESC ), [Delay Hours] )
CREATE MEASURE '_measures'[Reason Benchmark Gap Hours] = [Top 2 Controllable Reasons Hours] * [Opportunity Reduction %]
CREATE MEASURE '_measures'[Estimated Opportunity Tonnes] = DIVIDE ( [Reason Benchmark Gap Hours] * [Haul Truck Working Efficiency] * [Tonnes / Working h], 1000 )
CREATE MEASURE '_measures'[Unit Controllable Loss Rate] = DIVIDE ( [Controllable Delay Hours], [Calendar Hours] )
CREATE MEASURE '_measures'[Best Quartile Unit Controllable Loss Rate] = VAR _units = ALLSELECTED ( DimEquipment[UnitCode] ) RETURN PERCENTILEX.INC ( _units, CALCULATE ( [Unit Controllable Loss Rate] ), 0.25 )
CREATE MEASURE '_measures'[Unit Benchmark Gap Hours] = VAR _bq = [Best Quartile Unit Controllable Loss Rate] VAR _units = ALLSELECTED ( DimEquipment[UnitCode] ) RETURN SUMX ( _units, VAR _rate = CALCULATE ( [Unit Controllable Loss Rate] ) VAR _cal = CALCULATE ( [Calendar Hours] ) RETURN MAX ( 0, ( _rate - _bq ) * _cal ) )
CREATE MEASURE '_measures'[Opportunity vs Plan %] = DIVIDE ( [Estimated Opportunity Tonnes] * 1000, [Haulage Plan Tonnes] )
CREATE MEASURE '_measures'[Open WOs by Age Bucket] = VAR _asof = MAX ( DimDate[Date] ) + 1 VAR _bucket = SELECTEDVALUE ( BacklogAgeBucket[Bucket] ) RETURN CALCULATE ( COUNTROWS ( FactWorkOrder ), REMOVEFILTERS ( DimDate ), FILTER ( FactWorkOrder, VAR _age = DATEDIFF ( FactWorkOrder[StartTimestamp], _asof, DAY ) VAR _open = FactWorkOrder[StartTimestamp] < _asof && ( ISBLANK ( FactWorkOrder[CloseTimestamp] ) || FactWorkOrder[CloseTimestamp] >= _asof ) RETURN _open && SWITCH ( _bucket, "<7 d", _age < 7, "7-14 d", _age >= 7 && _age < 14, "14-30 d", _age >= 14 && _age < 30, "30-60 d", _age >= 30 && _age < 60, "60+ d", _age >= 60, FALSE () ) ) )
